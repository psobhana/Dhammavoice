𑀪𑀸𑀯𑀺𑀢 𑀓𑀭𑀦 𑀆𑀓𑀸𑀭𑀬

𑀧𑀴𑀫𑀼𑀯 + 𑀮𑀓𑀼𑀡 𑀫𑀢 𑀓𑁆𑀮𑀺𑀓𑁆 𑀓𑀭𑀦𑁆𑀦. 𑀦𑀺𑀮𑁆 𑀧𑀳𑀢𑀺 𑀘𑀢𑁆𑀬𑀬 𑀮𑀓𑀼𑀡 𑀫𑀢𑀼𑀯𑀦 𑀢𑀼𑀭𑀼 + 𑀮𑀓𑀼𑀡 𑀫𑀢 𑀓𑁆𑀮𑀺𑀓𑁆 𑀓𑀭𑀫𑀺𑀦𑁆 𑀇𑀤𑀺𑀭𑀺𑀬𑀝 𑀬𑀦𑁆𑀦. ''Loading'' 𑀤𑀼𑀝𑀼𑀯𑀳𑁄𑀢𑁆 𑀏𑀬 𑀦𑁄𑀲𑀮𑀓𑀸 𑀈𑀴 + 𑀮𑀓𑀼𑀡 𑀓𑁆𑀮𑀺𑀓𑁆 𑀓𑀭𑀫𑀺𑀦𑁆 𑀢𑀯𑀤𑀼𑀭𑀝𑀢𑁆 𑀇𑀤𑀺𑀭𑀺𑀬𑀝 𑀬𑀦𑁆𑀦. 𑀦𑀺𑀮𑁆 𑀧𑀳𑀢𑀺 𑀘𑀢𑁆𑀬 𑀲𑀳𑀺𑀢 (Blue Pagoda) 𑀫𑀸𑀢𑀓𑀸𑀯 𑀤𑀺𑀲𑁆 𑀯𑀽 𑀯𑀺𑀝 𑀏𑀬 𑀫𑀢 𑀓𑁆𑀮𑀺𑀓𑁆 𑀓𑀴 𑀯𑀺𑀝 YouTube 𑀓𑁄𑀝𑀼𑀯 𑀤𑀭𑁆𑀦𑀬 𑀯𑀦 𑀅𑀢𑀭 𑀅𑀯𑀲𑀸𑀦 𑀯𑀬𑁂𑀦𑁆 𑀏𑀫 𑀓𑁄𑀝𑀼𑀯 𑀫𑀢 𑀓𑁆𑀮𑀺𑀓𑁆 𑀓𑀴 𑀯𑀺𑀝 𑀩𑁆𑀤𑀬 𑁆𑀭𑀯𑀡𑀬 𑀓𑀴 𑀳𑀓𑀺𑀬.

𑀫𑁂𑀫 𑀲𑀚𑁆𑀛𑀸𑀬𑀦𑀸 𑀲𑀓𑀲𑁆 𑀓𑀺𑀭𑀻𑀫 𑀲𑀳𑀸 𑀓𑀢𑁆𑀭𑀺𑀫 𑀩𑀼𑀤𑁆𑀥𑀺𑀬 (AI) 𑀪𑀸𑀯𑀺𑀢 𑀓𑀭 𑀦𑁄𑀫𑀢.
